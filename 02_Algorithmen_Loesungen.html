static int zaehleLadebedarf(List<Fahrrad> raeder) {
    int anzahl = 0;                                       // Lücke 1

}

// a) alle einsatzbereiten Räder eines Typs mit Mindest-Akkustand in eine neue Liste
static List<Fahrrad> einsatzbereit(List<Fahrrad> raeder, String typ, int minAkku) {
    List<Fahrrad> ergebnis = new ArrayList<>();
    for (int i = 0; i < raeder.size(); i++) {
        Fahrrad r = raeder.get(i);
        if (r.getTyp().equals(typ) && !r.isDefekt() && r.getAkkuStand() >= minAkku) {
            ergebnis.add(r);
        }
    }
    return ergebnis;
}

// b) durchschnittliche Strecke aller Ausleihen, die an einer Station beginnen
static double durchschnittsStrecke(List<Ausleihe> ausleihen, int stationNr) {
    double summe = 0.0;
    int anzahl = 0;
    for (int i = 0; i < ausleihen.size(); i++) {
        Ausleihe a = ausleihen.get(i);
        if (a.getStartStation() == stationNr) {
            summe = summe + a.getStrecke();
            anzahl = anzahl + 1;
        }
    }
    if (anzahl == 0) {
        return 0.0;                 // keine Division durch 0
    }
    return summe / anzahl;
}

// a) Anzahl der Fahrten über 60 Minuten je Startstation; Index = stationNr - 1
static int[] langeFahrtenJeStation(List<Ausleihe> ausleihen, int anzahlStationen) {
    int[] ergebnis = new int[anzahlStationen];          // alle Elemente sind 0
    for (int i = 0; i < ausleihen.size(); i++) {
        Ausleihe a = ausleihen.get(i);
        if (a.getDauer() > 60) {
            int index = a.getStartStation() - 1;
            ergebnis[index] = ergebnis[index] + 1;
        }
    }
    return ergebnis;
}

// b) für jede Kundennummer genau ein KundenStatistik-Objekt (Werte noch 0)
static List<KundenStatistik> erzeugeStatistik(List<Ausleihe> ausleihen) {
    List<KundenStatistik> statistik = new ArrayList<>();
    for (int i = 0; i < ausleihen.size(); i++) {
        int kundenNr = ausleihen.get(i).getKundenNr();
        boolean vorhanden = false;
        int j = 0;
        while (j < statistik.size() && !vorhanden) {
            if (statistik.get(j).getKundenNr() == kundenNr) {
                vorhanden = true;
            }
            j = j + 1;
        }
        if (!vorhanden) {
            statistik.add(new KundenStatistik(kundenNr));
        }
    }
    return statistik;
}

// c) Werte je Kunde für den Zeitraum von..bis berechnen und setzen
static void berechneStatistik(List<Ausleihe> ausleihen, List<KundenStatistik> statistik,
                              Datum von, Datum bis) {
    for (int i = 0; i < statistik.size(); i++) {
        KundenStatistik ks = statistik.get(i);
        int anzahl = 0;                                   // für jeden Kunden neu!
        int summe = 0;
        for (int j = 0; j < ausleihen.size(); j++) {
            Ausleihe a = ausleihen.get(j);
            if (a.getKundenNr() == ks.getKundenNr()
                    && a.getDatum().liegtZwischen(von, bis)) {
                anzahl = anzahl + 1;
                summe = summe + a.getDauer();
            }
        }
        ks.setAnzahl(anzahl);
        ks.setGesamtDauer(summe);
        if (anzahl > 0) {
            ks.setDurchschnitt((double) summe / anzahl);   // keine Ganzzahldivision
        } else {
            ks.setDurchschnitt(0.0);
        }
    }
}

// a) Summe je Zeile (= je Wochentag)
static int[] summeJeTag(int[][] daten) {
    int[] summen = new int[daten.length];
    for (int tag = 0; tag < daten.length; tag++) {
        int summe = 0;
        for (int f = 0; f < daten[tag].length; f++) {
            summe = summe + daten[tag][f];
        }
        summen[tag] = summe;
    }
    return summen;
}

// b) Summe je Spalte (= je Zeitfenster)
static int[] summeJeZeitfenster(int[][] daten) {
    int anzahlFenster = daten[0].length;
    int[] summen = new int[anzahlFenster];
    for (int f = 0; f < anzahlFenster; f++) {
        int summe = 0;
        for (int tag = 0; tag < daten.length; tag++) {
            summe = summe + daten[tag][f];
        }
        summen[f] = summe;
    }
    return summen;
}

        throws KeineDatenException {

ALG-08 · Rekursion

Aufgabe ALG-08.1 · 4 Punkte

a) Der Basisfall ist die Abbruchbedingung: Für ihn liefert die Funktion ihr Ergebnis direkt, ohne sich selbst erneut aufzurufen (z. B. 0! = 1). Im Rekursionsfall ruft sich die Funktion mit einem kleineren bzw. einfacheren Teilproblem selbst auf und setzt aus dessen Ergebnis ihr eigenes zusammen (z. B. n! = n · (n − 1)!). Bewertung: a) je Begriff 1 P. b) Endlosrekursion 1 P, Stacküberlauf/Programmabbruch 1 P.

    String info = links <= rechts ? "  mitte = " + ((links + rechts) / 2) : "  -> links > rechts";
    System.out.println("   suche(nummern, " + gesucht + ", " + links + ", " + rechts + ")" + info);
    if (links > rechts) {
        return -1;                                      // Basisfall 1: Bereich leer

    if (nummern[mitte] == gesucht) {
        return mitte;                                   // Basisfall 2: gefunden

}

static int sucheIterativ(int[] nummern, int gesucht) {
    int links = 0;

            return mitte;
        } else if (nummern[mitte] < gesucht) {
            links = mitte + 1;
        } else {
            rechts = mitte - 1;
        }
    }
    return -1;
}

Aufgabe ALG-09.2 · 8 Punkte

a) Problem (4 P)

Es entsteht eine Doppelbuchung (Lost Update): Beide Threads lesen bei t1 und t2 den Wert 1, bevor einer von ihnen schreibt. Beide prüfen auf Basis dieses veralteten Werts (t3, t4), beide schreiben 0 und speichern eine Reservierung (t5, t6). Für ein einziges Lastenrad gibt es nun zwei Reservierungen; die Änderung von Thread A wird von B überschrieben, der Bestand müsste eigentlich „−1“ sein. Ursache: Die Folge Lesen – Prüfen – Schreiben ist ein kritischer Abschnitt, der nicht atomar bzw. nicht gegen gleichzeitigen Zugriff geschützt ist.

b) Pessimistisches Sperren (4 P)

Beim pessimistischen Sperren sperrt ein Thread den Datensatz vor dem Lesen (z. B. in einer Transaktion mit SELECT … FOR UPDATE, in Java mit synchronized/Lock). Andere Threads, die denselben Datensatz sperren wollen, müssen warten, bis die Sperre mit dem Ende der Transaktion (Commit) freigegeben wird. Dadurch kann immer nur ein Thread den kritischen Abschnitt durchlaufen.
